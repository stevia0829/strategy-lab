package main

import (
    "encoding/json"
    "fmt"
    "os"
    "strategylab/engine"
)

func main() {
    if len(os.Args) != 4 {
        fmt.Fprintln(os.Stderr, "expected strategy, CSV and output paths")
        os.Exit(2)
    }
    raw, err := os.ReadFile(os.Args[1])
    if err != nil { fail(err) }
    strategy, err := engine.ParseStrategy(raw)
    if err != nil { fail(err) }
    csv, err := os.ReadFile(os.Args[2])
    if err != nil { fail(err) }
    report, err := engine.RunWithTargets(strategy, string(csv), func(ctx engine.Context) ([]string, error) {
        return Decide(ctx), nil
    })
    if err != nil { fail(err) }
    result, err := json.Marshal(report)
    if err != nil { fail(err) }
    if err := os.WriteFile(os.Args[3], result, 0600); err != nil { fail(err) }
}

func fail(err error) {
    fmt.Fprintln(os.Stderr, err)
    os.Exit(2)
}
